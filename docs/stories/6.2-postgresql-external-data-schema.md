### AC1: PostgreSQL Schema Created ✅

Implement the following schema:

```sql
-- Metadata table for external data sources
CREATE TABLE external_data_sources (
    id SERIAL PRIMARY KEY,
    source_name VARCHAR(100) NOT NULL UNIQUE,  -- e.g., "INE_BuildingPermits"
    api_endpoint TEXT,
    data_type VARCHAR(50),  -- e.g., "time_series", "index"
    refresh_frequency VARCHAR(20),  -- e.g., "daily", "weekly", "monthly"
    last_refresh_at TIMESTAMP,
    created_at TIMESTAMP DEFAULT NOW(),
    metadata JSONB  -- Additional source-specific metadata
);

-- Time-series data points table
CREATE TABLE external_data_points (
    id SERIAL PRIMARY KEY,
    source_id INTEGER REFERENCES external_data_sources(id),
    date DATE NOT NULL,
    metric_name VARCHAR(100) NOT NULL,  -- e.g., "building_permits_count"
    value NUMERIC NOT NULL,
    unit VARCHAR(50),  -- e.g., "count", "EUR", "MWh"
    metadata JSONB,  -- Additional data-specific metadata
    created_at TIMESTAMP DEFAULT NOW(),
    UNIQUE(source_id, date, metric_name)
);

-- Indexes for fast queries
CREATE INDEX idx_data_points_source_date ON external_data_points(source_id, date);
CREATE INDEX idx_data_points_metric ON external_data_points(metric_name);
CREATE INDEX idx_sources_name ON external_data_sources(source_name);
```

```python
from sqlalchemy import Column, Integer, String, Numeric, Date, DateTime, ForeignKey, Index
from sqlalchemy.dialects.postgresql import JSONB
from sqlalchemy.orm import relationship
from raglite.shared.database import Base
from datetime import datetime

class ExternalDataSource(Base):
    """External data source metadata."""
    __tablename__ = "external_data_sources"

    id = Column(Integer, primary_key=True)
    source_name = Column(String(100), unique=True, nullable=False, index=True)
    api_endpoint = Column(String)
    data_type = Column(String(50))
    refresh_frequency = Column(String(20))
    last_refresh_at = Column(DateTime)
    created_at = Column(DateTime, default=datetime.utcnow)
    metadata = Column(JSONB, default={})

    # Relationship
    data_points = relationship("ExternalDataPoint", back_populates="source")

class ExternalDataPoint(Base):
    """Time-series data point from external source."""
    __tablename__ = "external_data_points"

    id = Column(Integer, primary_key=True)
    source_id = Column(Integer, ForeignKey("external_data_sources.id"), nullable=False)
    date = Column(Date, nullable=False)
    metric_name = Column(String(100), nullable=False)
    value = Column(Numeric, nullable=False)
    unit = Column(String(50))
    metadata = Column(JSONB, default={})
    created_at = Column(DateTime, default=datetime.utcnow)

    # Relationship
    source = relationship("ExternalDataSource", back_populates="data_points")

    # Indexes
    __table_args__ = (
        Index('idx_data_points_source_date', 'source_id', 'date'),
        Index('idx_data_points_metric', 'metric_name'),
        # Unique constraint
        UniqueConstraint('source_id', 'date', 'metric_name', name='uq_source_date_metric'),
    )
```

### Example: Data Storage Utility

```python
from sqlalchemy.orm import Session
from raglite.external_data.models import ExternalDataSource, ExternalDataPoint
from datetime import datetime, date
from typing import List, Optional

class ExternalDataStorage:
    """Utility for storing external data in PostgreSQL."""

    def __init__(self, session: Session):
        self.session = session

    def create_source(
        self,
        source_name: str,
        api_endpoint: str,
        data_type: str,
        refresh_frequency: str
    ) -> ExternalDataSource:
        """Create a new external data source."""
        source = ExternalDataSource(
            source_name=source_name,
            api_endpoint=api_endpoint,
            data_type=data_type,
            refresh_frequency=refresh_frequency
        )
        self.session.add(source)
        self.session.commit()
        return source

    def insert_data_points(
        self,
        source_name: str,
        data_points: List[dict]
    ) -> int:
        """Bulk insert data points for a source."""
        source = self.session.query(ExternalDataSource).filter_by(
            source_name=source_name
        ).first()

        if not source:
            raise ValueError(f"Source {source_name} not found")

        points = [
            ExternalDataPoint(
                source_id=source.id,
                date=dp["date"],
                metric_name=dp["metric_name"],
                value=dp["value"],
                unit=dp.get("unit"),
                metadata=dp.get("metadata", {})
            )
            for dp in data_points
        ]

        self.session.bulk_save_objects(points)
        self.session.commit()
        return len(points)

    def query_data_range(
        self,
        source_name: str,
        start_date: date,
        end_date: date,
        metric_name: Optional[str] = None
    ) -> List[ExternalDataPoint]:
        """Query data points for a date range."""
        query = self.session.query(ExternalDataPoint).join(
            ExternalDataSource
        ).filter(
            ExternalDataSource.source_name == source_name,
            ExternalDataPoint.date >= start_date,
            ExternalDataPoint.date <= end_date,
            ExternalDataPoint.deleted_at.is_(None)
        )

        if metric_name:
            query = query.filter(ExternalDataPoint.metric_name == metric_name)

        return query.order_by(ExternalDataPoint.date).all()
```
