### AC2: Automated Regressor Selection ✅

Implement correlation analysis to automatically select best regressors:

```python
def select_regressors(
    target: pd.Series,
    candidates: Dict[str, pd.Series],
    top_n: int = 7,
    min_correlation: float = 0.5
) -> List[str]:
    """
    Select top regressors based on Pearson correlation with target.

    Args:
        target: Target time series
        candidates: Dictionary of candidate regressors
        top_n: Maximum number of regressors to select
        min_correlation: Minimum correlation threshold

    Returns:
        List of selected regressor names
    """
    correlations = {}
    for name, series in candidates.items():
        # Align series and compute correlation
        aligned = pd.concat([target, series], axis=1).dropna()
        if len(aligned) > 10:  # Minimum data points
            corr = aligned.corr().iloc[0, 1]
            if abs(corr) >= min_correlation:
                correlations[name] = abs(corr)

    # Select top N by correlation
    selected = sorted(correlations.items(), key=lambda x: x[1], reverse=True)[:top_n]
    return [name for name, _ in selected]
```

**Fallback:**
- If correlation analysis fails, use manual selection
- Default regressors: building_permits, electricity_price, mortgage_loans

### AC3: Handle Missing Data ✅

Implement interpolation for missing regressor values:

```python
def prepare_regressors(
    regressors: Dict[str, pd.Series],
    target_index: pd.DatetimeIndex
) -> Dict[str, pd.Series]:
    """
    Prepare regressors by handling missing data.

    Args:
        regressors: Raw regressor data
        target_index: Target datetime index to align to

    Returns:
        Cleaned and aligned regressors
    """
    prepared = {}
    for name, series in regressors.items():
        # Reindex to target index
        aligned = series.reindex(target_index)

        # Check missing data percentage
        missing_pct = aligned.isna().sum() / len(aligned)
        if missing_pct > 0.10:
            raise ValueError(f"Regressor {name} has >10% missing data")

        # Interpolate missing values (linear)
        filled = aligned.interpolate(method='linear', limit=3)

        # Forward-fill remaining gaps (up to 3 periods)
        filled = filled.fillna(method='ffill', limit=3)

        if filled.isna().any():
            raise ValueError(f"Unable to fill all missing values in {name}")

        prepared[name] = filled

    return prepared
```

**Validation:**
- Raise error if >10% of regressor data missing
- Use linear interpolation for gaps <3 periods
- Forward-fill for edge cases (max 3 periods)

from prophet import Prophet
import pandas as pd
from typing import Dict, List, Optional
from raglite.external_data.storage import ExternalDataStorage

async def generate_forecast(
    metric: str,
    periods_ahead: int = 12,
    external_regressors: Optional[Dict[str, pd.Series]] = None,
    frequency: str = 'M'
) -> ForecastResult:
    """Generate forecast with optional external regressors."""

    # Fetch historical data for target metric
    historical = await fetch_historical_metric(metric)

    # Initialize Prophet model
    model = Prophet()

    # Add external regressors if provided
    if external_regressors:
        # Prepare regressors (handle missing data)
        prepared = prepare_regressors(external_regressors, historical.index)

        # Select top regressors by correlation
        selected = select_regressors(
            target=historical,
            candidates=prepared,
            top_n=7,
            min_correlation=0.5
        )

        logger.info(f"Selected regressors: {selected}")

        # Add to Prophet model
        for regressor_name in selected:
            model.add_regressor(regressor_name)

        # Prepare training data
        df = pd.DataFrame({
            'ds': historical.index,
            'y': historical.values
        })

        for regressor_name in selected:
            df[regressor_name] = prepared[regressor_name].values

        model_type = "prophet_multivariate"
        regressors_used = selected
    else:
        # Univariate forecasting (Epic 4 behavior)
        df = pd.DataFrame({
            'ds': historical.index,
            'y': historical.values
        })
        model_type = "prophet_univariate"
        regressors_used = []

    # Fit model
    model.fit(df)

    # Generate future dataframe
    future = model.make_future_dataframe(periods=periods_ahead, freq=frequency)

    # Add future regressor values (use latest available)
    if external_regressors and regressors_used:
        for regressor_name in regressors_used:
            latest_value = prepared[regressor_name].iloc[-1]
            future[regressor_name] = latest_value  # Assume constant

    # Predict
    forecast = model.predict(future)

    # Calculate accuracy metrics (if test data available)
    accuracy_metrics = calculate_accuracy(model, historical)

    # Calculate improvement vs baseline (if available)
    baseline_rmse = get_baseline_rmse(metric)  # Epic 4 univariate RMSE
    if baseline_rmse and external_regressors:
        improvement = (baseline_rmse - accuracy_metrics['rmse']) / baseline_rmse
    else:
        improvement = None

    return ForecastResult(
        metric=metric,
        predictions=[...],  # Convert forecast to ForecastPoint objects
        model_type=model_type,
        accuracy_metrics=accuracy_metrics,
        regressors_used=regressors_used,
        improvement_vs_baseline=improvement
    )
